Lesson 0019 · Pion·WebRTC

SFU — 1:N 방송의 구조

LiveKit의 정체가 바로 SFU다. 오늘은 그 최소 동작 원리 — 발행자 하나의 RTP를 재인코딩 없이 여러 구독자에게 복제 전달 — 를 pion으로 직접 만들어 실측으로 증명한다. 예상 소요: 45–50분. (pion/webrtc v4.2.18로 검증)

오늘의 목표 (하나만)

한 프로세스 안에 발행자 1 + 미니 SFU + 구독자 2를 세우고, 같은 RTP 패킷 열이 두 구독자에게 복제 전달되는 것(seq 통과, SSRC는 연결별 재작성)을 실측 로그로 설명할 수 있다.

1. 왜 mesh로는 안 되는가 — 업링크 수학

참가자 N명이 서로 직접 연결(full mesh)하면 각자 N−1개의 스트림을 업로드해야 한다. 5인 통화면 내 업링크로 4개, 10인이면 9개 — 모바일 업링크는 몇 Mbps가 한계라 금방 무너진다. 해법은 서버를 끼우는 것이고, 방식이 둘이다 (WebRTC for the Curious — Applied WebRTC):

SFU (Selective Forwarding Unit)MCU (Multipoint Conferencing Unit)
서버가 하는 일RTP 패킷을 선별해서 그대로 전달모든 스트림을 디코딩 → 한 화면으로 합성 → 재인코딩
서버 CPU낮음 (인코딩 없음)매우 높음 (참가자 수만큼 코덱 연산)
지연낮음합성·재인코딩만큼 추가
클라이언트업로드 1개, 다운로드 N−1개 (레이아웃 자유)업로드 1개, 다운로드 1개 (고정 화면)

미디어 서버 업계의 기본값은 SFU다 — LiveKit도, mediamtx의 WebRTC 배포도 이 모델이다. 발행자는 한 번만 업로드하고, 서버가 팬아웃을 담당한다.

2. SFU의 핵심 동작 세 가지

  • 재인코딩 없는 전달 — 압축된 RTP 페이로드를 열지 않고 그대로 흘린다. 서버는 코덱을 몰라도 된다.
  • PLI (Picture Loss Indication) — 새 구독자는 키프레임부터 받아야 화면을 그릴 수 있다. SFU는 RTCP 피드백 PLI를 발행자 쪽으로 보내 키프레임 생성을 요청한다. pion 공식 broadcast 예제는 intervalpli 인터셉터로 3초마다 PLI를 보낸다 — "A PLI causes a video keyframe to be generated by the sender" (예제 주석 원문).
  • NACK — 구독자가 잃어버린 seq를 RTCP로 신고하면 재전송한다 (개념만 — 인터셉터가 자동 처리하며, 다음 레슨에서 다룬다).

3. pion으로 미니 SFU 설계

발행자 pub ──PC── [ sfuIn │ 미니 SFU │ sfuOut2 ──PC── 구독자 sub2 ]
                  [        │          │ sfuOut3 ──PC── 구독자 sub3 ]

SFU는 클라이언트당 PeerConnection을 하나씩 갖는다 — 우리 미니어처도 3개(인그레스 1 + 이그레스 2), LiveKit도 참가자당 하나다. 중계 파이프는 두 API로 만든다:

  • 인그레스: OnTrack으로 받은 TrackRemote에서 ReadRTP() — 레슨 0018에서 본 그 수신 루프
  • 이그레스: TrackLocalStaticRTP에 WriteRTP() — "Static"은 seq·timestamp를 손대지 않고 통과시킨다는 뜻이고, SSRC·페이로드 타입은 연결별 협상값으로 pion이 재작성한다

4. 실습 — 1→2 복제 전달을 실측하기

  1. playground/webrtc-sfu를 만들고 go mod init webrtcsfu 후 go get github.com/pion/webrtc/v4
  2. main.go에 먼저 시그널링 헬퍼와 구독자 통계 — 레슨 0015의 non-trickle 교환 그대로: 
    package main

import (
	"fmt"
	"sync"
	"sync/atomic"
	"time"

	"github.com/pion/webrtc/v4"
	"github.com/pion/webrtc/v4/pkg/media"
)

// connect는 두 PeerConnection을 non-trickle offer/answer로 연결한다.
func connect(offerer, answerer *webrtc.PeerConnection) error {
	offer, err := offerer.CreateOffer(nil)
	if err != nil {
		return fmt.Errorf("offer 생성: %w", err)
	}
	offerGathered := webrtc.GatheringCompletePromise(offerer)
	if err := offerer.SetLocalDescription(offer); err != nil {
		return err
	}
	<-offerGathered
	if err := answerer.SetRemoteDescription(*offerer.LocalDescription()); err != nil {
		return err
	}
	answer, err := answerer.CreateAnswer(nil)
	if err != nil {
		return err
	}
	answerGathered := webrtc.GatheringCompletePromise(answerer)
	if err := answerer.SetLocalDescription(answer); err != nil {
		return err
	}
	<-answerGathered
	return offerer.SetRemoteDescription(*answerer.LocalDescription())
}

// subStats는 구독자 한 명이 받은 RTP 스트림의 실측치를 모은다.
type subStats struct {
	mu          sync.Mutex
	ssrc        uint32
	count       int
	first, last uint16
}

func (s *subStats) record(ssrc uint32, seq uint16) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.count == 0 {
		s.ssrc = ssrc
		s.first = seq
	}
	s.count++
	s.last = seq
}
  3. RTCP 소비 고루틴과 구독자 생성 — 0012의 뮤텍스, 0007의 고루틴이 그대로 실전이 된다: 
    // drainRTCP는 송신 트랙의 RTCP를 소비한다 — 안 읽으면 인터셉터 피드백이 멈춘다.
func drainRTCP(sender *webrtc.RTPSender, label string, report bool) {
	go func() {
		types := map[string]int{}
		for {
			pkts, _, err := sender.ReadRTCP()
			if err != nil {
				if report {
					fmt.Printf("[rtcp:%s] 수신 요약: %v (종료: %v)\n", label, types, err)
				}
				return
			}
			for _, p := range pkts {
				types[fmt.Sprintf("%T", p)]++
			}
		}
	}()
}

// newSubscriber는 구독자 PC를 만들고 OnTrack에서 RTP를 세는 고루틴을 건다.
func newSubscriber(name string, stats *subStats) (*webrtc.PeerConnection, error) {
	pc, err := webrtc.NewPeerConnection(webrtc.Configuration{})
	if err != nil {
		return nil, err
	}
	pc.OnTrack(func(track *webrtc.TrackRemote, _ *webrtc.RTPReceiver) {
		fmt.Printf("[%s] OnTrack: codec=%s ssrc=%d\n", name, track.Codec().MimeType, track.SSRC())
		for {
			pkt, _, err := track.ReadRTP()
			if err != nil {
				fmt.Printf("[%s] ReadRTP 종료: %v\n", name, err)
				return
			}
			stats.record(uint32(track.SSRC()), pkt.SequenceNumber)
		}
	})
	return pc, nil
}
  4. main — 6개 PC를 만들고, SFU 중계 루프를 걸고, 연결 후 100 샘플을 발행한다: 
    func main() {
	// ── 발행자와 SFU 인그레스 ─────────────────────────────
	pub, err := webrtc.NewPeerConnection(webrtc.Configuration{})
	if err != nil {
		panic(err)
	}
	sfuIn, err := webrtc.NewPeerConnection(webrtc.Configuration{})
	if err != nil {
		panic(err)
	}

	pubTrack, err := webrtc.NewTrackLocalStaticSample(
		webrtc.RTPCodecCapability{MimeType: webrtc.MimeTypeVP8}, "video", "publisher")
	if err != nil {
		panic(err)
	}
	pubSender, err := pub.AddTrack(pubTrack)
	if err != nil {
		panic(err)
	}
	drainRTCP(pubSender, "발행자", true)

	// ── SFU 이그레스(구독자별 PC 1개)와 구독자들 ─────────────
	var stats2, stats3 subStats

	fwd2, err := webrtc.NewTrackLocalStaticRTP(
		webrtc.RTPCodecCapability{MimeType: webrtc.MimeTypeVP8}, "video", "sfu")
	if err != nil {
		panic(err)
	}
	fwd3, err := webrtc.NewTrackLocalStaticRTP(
		webrtc.RTPCodecCapability{MimeType: webrtc.MimeTypeVP8}, "video", "sfu")
	if err != nil {
		panic(err)
	}

	sfuOut2, err := webrtc.NewPeerConnection(webrtc.Configuration{})
	if err != nil {
		panic(err)
	}
	sfuOut3, err := webrtc.NewPeerConnection(webrtc.Configuration{})
	if err != nil {
		panic(err)
	}
	s2, err := sfuOut2.AddTrack(fwd2)
	if err != nil {
		panic(err)
	}
	s3, err := sfuOut3.AddTrack(fwd3)
	if err != nil {
		panic(err)
	}
	drainRTCP(s2, "sfu→sub2", false)
	drainRTCP(s3, "sfu→sub3", false)

	sub2, err := newSubscriber("sub2", &stats2)
	if err != nil {
		panic(err)
	}
	sub3, err := newSubscriber("sub3", &stats3)
	if err != nil {
		panic(err)
	}

	// ── SFU 핵심: 발행자 RTP를 두 구독자 트랙으로 복제 전달 ────
	var forwarded atomic.Int64
	sfuDone := make(chan struct{})
	sfuIn.OnTrack(func(track *webrtc.TrackRemote, _ *webrtc.RTPReceiver) {
		fmt.Printf("[sfu] 발행자 트랙 수신: ssrc=%d pt=%d\n", track.SSRC(), track.PayloadType())
		defer close(sfuDone)
		for {
			pkt, _, err := track.ReadRTP()
			if err != nil {
				fmt.Printf("[sfu] 발행자 ReadRTP 종료: %v\n", err)
				return
			}
			if err := fwd2.WriteRTP(pkt); err != nil {
				fmt.Printf("[sfu] fwd2 쓰기 실패: %v\n", err)
			}
			if err := fwd3.WriteRTP(pkt); err != nil {
				fmt.Printf("[sfu] fwd3 쓰기 실패: %v\n", err)
			}
			forwarded.Add(1)
		}
	})

	// ── 세 연결 수립 ─────────────────────────────────────
	for _, pair := range []struct {
		name string
		a, b *webrtc.PeerConnection
	}{
		{"pub↔sfu", pub, sfuIn},
		{"sfu↔sub2", sfuOut2, sub2},
		{"sfu↔sub3", sfuOut3, sub3},
	} {
		if err := connect(pair.a, pair.b); err != nil {
			panic(fmt.Errorf("%s 연결 실패: %w", pair.name, err))
		}
		fmt.Printf("[시그널링] %s offer/answer 완료\n", pair.name)
	}

	// ── 발행: 20ms 간격으로 100 샘플 (약 2초 분량) ───────────
	payload := make([]byte, 600)
	for i := 0; i < 100; i++ {
		if err := pubTrack.WriteSample(media.Sample{Data: payload, Duration: 20 * time.Millisecond}); err != nil {
			panic(err)
		}
		time.Sleep(20 * time.Millisecond)
	}
	time.Sleep(500 * time.Millisecond) // 마지막 패킷이 구독자에 닿을 시간

	// ── 실측 요약 ────────────────────────────────────────
	fmt.Printf("[sfu] 중계한 RTP 패킷: %d개\n", forwarded.Load())
	for _, sub := range []struct {
		name string
		s    *subStats
	}{{"sub2", &stats2}, {"sub3", &stats3}} {
		sub.s.mu.Lock()
		fmt.Printf("[%s] 수신 %d개, ssrc=%d, seq %d→%d\n", sub.name, sub.s.count, sub.s.ssrc, sub.s.first, sub.s.last)
		sub.s.mu.Unlock()
	}

	// ── 부숴보기(b): 발행자가 끊기면 SFU의 ReadRTP는? ────────
	fmt.Println("--- 발행자 Close() ---")
	if err := pub.Close(); err != nil {
		panic(err)
	}
	select {
	case <-sfuDone:
	case <-time.After(3 * time.Second):
		fmt.Println("[sfu] 3초 내 ReadRTP 종료 감지 못함")
	}

	for _, pc := range []*webrtc.PeerConnection{sfuIn, sfuOut2, sfuOut3, sub2, sub3} {
		_ = pc.Close()
	}
	time.Sleep(300 * time.Millisecond) // 리더 고루틴들의 종료 로그 대기
}

실측 읽기 — 이 로그가 증명하는 것

  • SSRC 세 개가 전부 다르다 (발행자 3063034383, sub2 3692589650, sub3 2148735737) — SSRC는 연결(leg)마다 협상되고, TrackLocalStaticRTP가 자동 재작성한다. 레슨 0018의 "SSRC = 스트림 식별자"가 SFU를 건너면 이렇게 된다.
  • seq 범위는 두 구독자가 동일하다 (→31704) — "Static" = seq·timestamp 통과. 같은 패킷 열의 복제 전달이라는 증거.
  • 수신 수가 99/98/97로 조금씩 다르다 — 시작 직후 몇 패킷은 구독자 파이프가 준비되기 전에 도착해 버려진다. RTP는 손실 허용 설계라 이게 정상이고, 실전에서 그 공백을 메우는 게 §2의 PLI다.
  • seq가 31607부터 시작한다 — RTP 시퀀스는 보안상 무작위 값에서 시작한다 (RFC 3550). 0부터가 아니다.

부숴보기 — 이번엔 "조용한 문제" 두 가지

  1. drainRTCP(...) 세 줄을 지우고 실행해 보라.

    실측: 3초간 아무 증상 없음 — 수신 149/150개, 에러·경고 0건. 그런데 발행자 sender에는 2.5초 동안 ReceiverReport가 2개 도착하고 있었다(위 로그의 map[*rtcp.ReceiverReport:2]). 안 읽으면 NACK·수신 리포트가 인터셉터에 도달하지 못해 손실 복구와 통계가 조용히 죽는다 — 짧은 데모에선 안 보이는, 장수 SFU에서 터지는 종류의 버그다. 공식 broadcast 예제가 sender마다 RTCP 소비 고루틴을 태우는 이유. 확인 후 복구.

  2. 이미 프로그램에 들어 있는 부숴보기: pub.Close() 후 로그를 다시 보라.

    SFU 인그레스는 ReadRTP 종료: EOF로 즉시 안다. 하지만 구독자들은 에러 없이 계속 기다린다 — 패킷이 안 올 뿐이다 (마지막 두 줄의 EOF는 우리가 구독자 PC를 닫아서 난 것). 발행자 이탈을 구독자에게 전파하는 건 SFU의 책임이며, LiveKit의 participant 라이프사이클 관리가 정확히 이 일이다.
