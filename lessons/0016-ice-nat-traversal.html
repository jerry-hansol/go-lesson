Lesson 0016 · Pion·WebRTC

ICE — NAT를 뚫는 법

레슨 0015에서 두 피어가 SDP를 교환했다. 그런데 그 패킷은 실제로 어디로 가는가? 인터넷의 대부분은 NAT 뒤에 있다 — 오늘은 WebRTC가 그 벽을 뚫는 프레임워크, ICE를 pion으로 실측한다. 예상 소요: 40–45분.

오늘의 목표 (하나만)

pion으로 ICE candidate를 실물 관찰하고(host → STUN을 켜면 srflx), 연결 검사(checking→connected)와 선택된 candidate pair를 직접 확인한다 — 그리고 ICE가 시그널링 유실마저 prflx로 살려내는 것을 목격한다.

1. 왜 P2P가 그냥 안 되는가

백엔드 서버는 공인 IP(또는 LB) 뒤에서 기다리는 쪽이라 NAT가 문제되지 않는다. P2P는 다르다 — 양쪽 모두 사설 IP 뒤에 있고, 서로의 NAT는 "안에서 먼저 나간 적 있는 매핑"으로만 인바운드를 통과시킨다. 그래서 WebRTC는 연결 수립만을 위한 전용 서브시스템을 갖는다: ICE(Interactive Connectivity Establishment)다 (WebRTC for the Curious — Connecting).

2. Candidate — "내가 도달 가능할지도 모르는 주소 후보들"

ICE의 재료는 candidate, 즉 자신이 패킷을 받을 수 있을 것 같은 주소:포트 후보다. 양쪽이 후보를 모아 시그널링으로 교환하고, 모든 조합(candidate pair)에 STUN 바인딩 요청을 서로 쏘아 실제로 통하는 pair를 찾는다 — 이것이 연결 검사(connectivity check)고, 그동안 ICE 상태는 checking이다.

타입정체어디서 오나
host내 NIC의 주소 그대로로컬 인터페이스 열거
srflxNAT 밖에서 보이는 내 공인 주소STUN 서버의 응답
prflx연결 검사 도중 역발견된 상대 주소수신된 검사 패킷의 출발지
relay중계 서버의 주소 (최후의 수단)TURN 서버 할당

3. STUN vs TURN, 그리고 trickle

  • STUN은 거울이다 — "밖에서 너는 이렇게 보여"라고 공인 주소:포트를 알려줄 뿐, 트래픽을 나르지 않는다. 운영 비용이 거의 없다.
  • TURN은 릴레이다 — 양쪽 NAT 타입이 호환되지 않아 직결이 불가능할 때 미디어가 서버를 경유한다. 스트림 대역폭 전체가 서버 비용이 된다 — 미디어 서비스에서 TURN 비율은 곧 인프라 비용이다.
  • trickle ICE: 수집이 다 끝나길 기다리지 않고 candidate가 모이는 대로 상대에게 흘려보내 연결 시간을 줄인다. 아래 실습의 OnICECandidate → AddICECandidate 중계가 정확히 그것이다.

4. 실습 — candidate 실물 관찰과 연결 검사

레슨 0008의 레이아웃대로 한 모듈에 프로그램 두 개를 둔다. pion은 v4를 쓴다 (이 레슨의 실측: v4.2.18).

  1. 모듈 준비: 
    mkdir -p ~/Desktop/repo/learn-go/playground/webrtc-ice
cd ~/Desktop/repo/learn-go/playground/webrtc-ice
go mod init webrtcice
go get github.com/pion/webrtc/v4
  2. gather/main.go — candidate 관찰기: 
    package main

import (
	"fmt"
	"time"

	"github.com/pion/webrtc/v4"
)

// useStun을 true로 바꾸면 STUN 서버가 알려준 srflx candidate가 추가로 나온다 (3단계).
const useStun = false

// must는 예제 축약용 — 실무 코드라면 error를 반환한다 (레슨 0003).
func must(err error) {
	if err != nil {
		panic(err)
	}
}

func main() {
	config := webrtc.Configuration{}
	if useStun {
		config.ICEServers = []webrtc.ICEServer{
			{URLs: []string{"stun:stun.l.google.com:19302"}},
		}
	}

	pc, err := webrtc.NewPeerConnection(config)
	must(err)
	defer pc.Close()

	// candidate가 하나 모일 때마다 호출된다 — trickle ICE의 재료
	pc.OnICECandidate(func(c *webrtc.ICECandidate) {
		if c == nil { // nil = 수집 종료 신호
			fmt.Println("수집 완료")
			return
		}
		fmt.Printf("candidate: typ=%-6s proto=%s %s:%d\n", c.Typ, c.Protocol, c.Address, c.Port)
	})

	// 협상할 것이 있어야 수집이 시작된다 — 데이터 채널 하나 추가
	_, err = pc.CreateDataChannel("probe", nil)
	must(err)

	offer, err := pc.CreateOffer(nil)
	must(err)
	must(pc.SetLocalDescription(offer)) // SetLocalDescription이 ICE 수집의 방아쇠

	time.Sleep(3 * time.Second)
}

    go run ./gather 실측 (주소 일부는 가렸다):
    candidate: typ=host proto=udp 192.168.1.78:56687
    candidate: typ=host proto=udp 2001:…:1ee8:52932 (IPv6 ×2)
    candidate: typ=host proto=udp 192.168.254.16:50319
    수집 완료 — ICE 서버가 없으니 전부 host다.

  3. useStun = true로 바꾸고 다시 실행:

    실측: host 4개에 이어 candidate: typ=srflx proto=udp 64.180.…:50873 — STUN이 알려준 공인 IPv4가 srflx로 등장한다. 사내망 등에서 UDP가 막혀 있으면 srflx가 안 나올 수 있는데, 그 부재 자체가 "이 네트워크는 STUN 불가"라는 진단 신호다.

  4. connect/main.go — 두 피어를 한 프로세스에서 연결하고 상태 전이를 관찰: 
    package main

import (
	"fmt"
	"time"

	"github.com/pion/webrtc/v4"
)

// oneWay를 true로 바꾸면 pc2→pc1 방향 candidate 전달을 끊는다 (부숴보기 a).
const oneWay = false

// must는 예제 축약용 — 실무 코드라면 error를 반환한다 (레슨 0003).
func must(err error) {
	if err != nil {
		panic(err)
	}
}

func main() {
	pc1, err := webrtc.NewPeerConnection(webrtc.Configuration{})
	must(err)
	defer pc1.Close()
	pc2, err := webrtc.NewPeerConnection(webrtc.Configuration{})
	must(err)
	defer pc2.Close()

	// trickle ICE: 모이는 candidate를 즉시 상대에게 전달 (실전에선 시그널링 서버가 중계)
	pc1.OnICECandidate(func(c *webrtc.ICECandidate) {
		if c != nil {
			must(pc2.AddICECandidate(c.ToJSON()))
		}
	})
	pc2.OnICECandidate(func(c *webrtc.ICECandidate) {
		if c != nil && !oneWay {
			must(pc1.AddICECandidate(c.ToJSON()))
		}
	})

	pc1.OnICEConnectionStateChange(func(s webrtc.ICEConnectionState) { fmt.Println("pc1 ICE:", s) })
	pc2.OnICEConnectionStateChange(func(s webrtc.ICEConnectionState) { fmt.Println("pc2 ICE:", s) })

	dc, err := pc1.CreateDataChannel("probe", nil)
	must(err)
	done := make(chan struct{})
	dc.OnOpen(func() {
		fmt.Println("데이터 채널 열림 — 연결 성립")
		close(done)
	})

	// offer/answer 교환 (레슨 0015) — SetLocalDescription 시점부터 수집·trickle 시작
	offer, err := pc1.CreateOffer(nil)
	must(err)
	must(pc1.SetLocalDescription(offer))
	must(pc2.SetRemoteDescription(offer))
	answer, err := pc2.CreateAnswer(nil)
	must(err)
	must(pc2.SetLocalDescription(answer))
	must(pc1.SetRemoteDescription(answer))

	select {
	case <-done:
		printSelectedPair("pc1", pc1)
		printSelectedPair("pc2", pc2)
	case <-time.After(10 * time.Second):
		fmt.Println("10초 안에 연결 실패 — ICE 상태를 보라")
	}
}

// printSelectedPair는 실제 통신에 선택된 candidate pair의 타입을 보여준다.
func printSelectedPair(name string, pc *webrtc.PeerConnection) {
	pair, err := pc.SCTP().Transport().ICETransport().GetSelectedCandidatePair()
	if err != nil || pair == nil {
		return
	}
	fmt.Printf("%s 선택된 pair: local=%s ↔ remote=%s\n", name, pair.Local.Typ, pair.Remote.Typ)
}

    go run ./connect 실측 (2회 연속 동일):
    pc1 ICE: checking → pc2 ICE: checking → 양쪽 connected → 데이터 채널 열림 — 연결 성립
    pc1 선택된 pair: local=host ↔ remote=host — 같은 머신이니 host끼리 붙는 게 정상이다.

부숴보기 — 그런데 ICE는 잘 안 부서진다

  1. oneWay = true로 바꿔 pc2→pc1 candidate 전달을 끊고 실행:

    부서질 것 같지만 — 실측: 연결이 성립한다. pc1 선택된 pair: local=host ↔ remote=prflx. pc1은 pc2의 candidate를 하나도 못 받았지만, pc2가 보낸 연결 검사 패킷의 출발지 주소에서 상대를 역발견해 prflx candidate를 만들었다 (원전의 Peer Reflexive 설명). 단, 이건 검사 패킷이 물리적으로 도달 가능했기 때문 — 대칭 NAT 너머 실전에서는 양방향 교환과 TURN 폴백이 여전히 필수다. 확인 후 원상복구.

  2. STUN 주소를 stun:no-such-host.invalid:19302로 바꿔 gather 실행:

    실측: 에러도 panic도 없이 host candidate만 나오고 수집 완료. STUN 실패는 조용하다 — srflx가 안 보이면 오타·차단부터 의심하라. 삼켜진 에러는 pion 로거로 발굴한다:
    PION_LOG_DEBUG=all go run ./gather → ice DEBUG: gather.go:861: Failed to resolve STUN host: … no such host (실측). 확인 후 원상복구.
