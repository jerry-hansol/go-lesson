캡스톤 — 미니 스트림 게이트웨이

새 개념은 없습니다. 레슨 0002~0012의 전부를 하나의 동작하는 서비스로 조립합니다 — HTTP API, 뮤텍스 보호 저장소, 채널 기반 워커, graceful shutdown, -race 통과 테스트까지. 예상 소요: 60–90분.

오늘의 목표 (하나만)

지금까지 배운 개념만으로 스트림 등록·조회 API와 백그라운드 세그먼트 워커를 갖춘 게이트웨이를 완성하고, go vet·go test -race·graceful shutdown 3종 검증을 통과시킨다. 이게 끝나면 기초·실전 아크 졸업 — 다음은 진짜 코드베이스(mediamtx) 읽기다.

  2. internal/store/store.go — 뮤텍스로 보호되는 저장소. 핵심 결정: 잠금 밖으로는 내부 포인터가 아니라 복사본을 내보낸다 (레슨 0012의 교훈): 
    package store

import (
	"fmt"
	"sync"
)

// Stream은 게이트웨이에 등록된 스트림 하나의 상태다.
type Stream struct {
	ID          string `json:"id"`
	Name        string `json:"name"`
	BitrateKbps int    `json:"bitrate_kbps"`
	Segments    int    `json:"segments"` // 워커가 처리 완료한 세그먼트 수
}

// Store는 뮤텍스로 보호되는 인메모리 스트림 저장소다.
// 모든 메서드가 잠금을 잡으므로 여러 고루틴에서 동시에 써도 안전하다.
type Store struct {
	mu      sync.Mutex
	streams map[string]*Stream
	nextID  int
}

// New는 바로 사용 가능한 빈 Store를 만든다.
func New() *Store {
	return &Store{streams: make(map[string]*Stream)}
}

// Add는 스트림을 등록하고 ID가 부여된 복사본을 반환한다.
func (s *Store) Add(name string, bitrateKbps int) Stream {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.nextID++
	st := &Stream{
		ID:          fmt.Sprintf("s%d", s.nextID),
		Name:        name,
		BitrateKbps: bitrateKbps,
	}
	s.streams[st.ID] = st
	return *st
}

// Get은 ID로 스트림을 찾는다. 없으면 ok=false를 돌려준다.
// 잠금 밖에서 안전하도록 내부 포인터가 아닌 복사본을 반환한다.
func (s *Store) Get(id string) (Stream, bool) {
	s.mu.Lock()
	defer s.mu.Unlock()
	st, ok := s.streams[id]
	if !ok {
		return Stream{}, false
	}
	return *st, true
}

// List는 등록된 모든 스트림의 복사본을 반환한다. 순서는 정해져 있지 않다.
func (s *Store) List() []Stream {
	s.mu.Lock()
	defer s.mu.Unlock()
	out := make([]Stream, 0, len(s.streams))
	for _, st := range s.streams {
		out = append(out, *st)
	}
	return out
}

// AddSegment는 워커가 세그먼트 하나를 처리 완료했음을 기록한다.
func (s *Store) AddSegment(id string) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if st, ok := s.streams[id]; ok {
		st.Segments++
	}
}

// Totals는 (스트림 수, 처리 완료 세그먼트 합계)를 반환한다.
func (s *Store) Totals() (int, int) {
	s.mu.Lock()
	defer s.mu.Unlock()
	segments := 0
	for _, st := range s.streams {
		segments += st.Segments
	}
	return len(s.streams), segments
}

  3. internal/store/store_test.go — 테이블 드리븐 + 동시성 테스트 (레슨 0009): 
    package store

import (
	"sync"
	"testing"
)

// TestGet은 존재/부재 케이스를 테이블로 검증한다.
func TestGet(t *testing.T) {
	s := New()
	created := s.Add("cam-1", 2500)

	tests := []struct {
		name   string
		id     string
		wantOK bool
	}{
		{"존재하는 ID", created.ID, true},
		{"없는 ID", "s999", false},
		{"빈 ID", "", false},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			_, ok := s.Get(tt.id)
			if ok != tt.wantOK {
				t.Errorf("Get(%q) ok = %v, want %v", tt.id, ok, tt.wantOK)
			}
		})
	}
}

// TestAdd는 ID 부여와 필드 보존을 검증한다.
func TestAdd(t *testing.T) {
	s := New()
	first := s.Add("cam-1", 2500)
	second := s.Add("cam-2", 4000)

	if first.ID == second.ID {
		t.Errorf("ID가 중복됨: %q", first.ID)
	}
	got, ok := s.Get(second.ID)
	if !ok || got.Name != "cam-2" || got.BitrateKbps != 4000 {
		t.Errorf("Get(%q) = %+v, ok=%v", second.ID, got, ok)
	}
}

// TestConcurrentSegments는 고루틴 100개가 동시에 세그먼트를 기록해도
// 유실이 없는지 검증한다. -race 플래그의 주 타깃이다.
func TestConcurrentSegments(t *testing.T) {
	s := New()
	st := s.Add("cam-1", 2500)

	var wg sync.WaitGroup
	for i := 0; i < 100; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			s.AddSegment(st.ID)
		}()
	}
	wg.Wait()

	got, _ := s.Get(st.ID)
	if got.Segments != 100 {
		t.Errorf("Segments = %d, want 100", got.Segments)
	}
}

  4. internal/worker/worker.go — 채널에서 작업을 받아 처리하는 워커 풀. select가 종료 신호와 작업을 동시에 기다린다 (0007 + 0011): 
    package worker

import (
	"context"
	"log"
	"sync"
	"time"

	"gateway/internal/store"
)

// Job은 워커가 처리할 세그먼트 작업 하나다.
type Job struct {
	StreamID string
	Segment  int
}

// Pool은 공용 채널에서 Job을 받아 처리하는 워커 고루틴 집합이다.
type Pool struct {
	Jobs chan Job // 핸들러가 여기로 작업을 보낸다
	st   *store.Store
	wg   sync.WaitGroup
}

// New는 워커 n개를 시작해 Pool을 반환한다. ctx가 취소되면 워커들이 멈춘다.
func New(ctx context.Context, st *store.Store, n int) *Pool {
	p := &Pool{Jobs: make(chan Job, 64), st: st}
	for i := 1; i <= n; i++ {
		p.wg.Add(1)
		go p.run(ctx, i)
	}
	return p
}

// run은 워커 하나의 본체 — 종료 신호와 작업 채널을 select로 동시에 기다린다.
func (p *Pool) run(ctx context.Context, id int) {
	defer p.wg.Done()
	for {
		select {
		case <-ctx.Done(): // graceful shutdown 신호
			log.Printf("worker %d: 종료", id)
			return
		case job := <-p.Jobs:
			time.Sleep(50 * time.Millisecond) // 트랜스코딩 흉내
			p.st.AddSegment(job.StreamID)
			log.Printf("worker %d: %s seg-%d 처리 완료", id, job.StreamID, job.Segment)
		}
	}
}

// Wait는 모든 워커 고루틴이 끝날 때까지 블록한다.
func (p *Pool) Wait() { p.wg.Wait() }

  5. cmd/gateway/main.go — 전부 조립. 새 API는 signal.NotifyContext 하나뿐이며 주석으로 설명했다 (pkg.go.dev/os/signal): 
    package main

import (
	"context"
	"encoding/json"
	"errors"
	"log"
	"net/http"
	"os/signal"
	"syscall"
	"time"

	"gateway/internal/store"
	"gateway/internal/worker"
)

// createRequest는 POST /streams의 요청 본문이다.
type createRequest struct {
	Name        string `json:"name"`
	BitrateKbps int    `json:"bitrate_kbps"`
	Segments    int    `json:"segments"`
}

func main() {
	// signal.NotifyContext: Ctrl-C(SIGINT)나 SIGTERM이 오면
	// 자동으로 취소되는 context를 만든다 — 종료 신호를 context 세계로 번역.
	ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
	defer stop()

	st := store.New()
	pool := worker.New(ctx, st, 3)

	mux := http.NewServeMux()

	// 스트림 등록 → 세그먼트 작업을 워커 채널로 전달
	mux.HandleFunc("POST /streams", func(w http.ResponseWriter, r *http.Request) {
		var req createRequest
		if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
			http.Error(w, "잘못된 JSON: "+err.Error(), http.StatusBadRequest)
			return
		}
		if req.Name == "" || req.Segments <= 0 {
			http.Error(w, "name과 1 이상의 segments가 필요합니다", http.StatusBadRequest)
			return
		}
		created := st.Add(req.Name, req.BitrateKbps)
		for i := 1; i <= req.Segments; i++ {
			pool.Jobs <- worker.Job{StreamID: created.ID, Segment: i}
		}
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusCreated)
		json.NewEncoder(w).Encode(created)
	})

	// 전체 목록
	mux.HandleFunc("GET /streams", func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		json.NewEncoder(w).Encode(st.List())
	})

	// 단건 조회 — 없으면 404
	mux.HandleFunc("GET /streams/{id}", func(w http.ResponseWriter, r *http.Request) {
		s, ok := st.Get(r.PathValue("id"))
		if !ok {
			http.Error(w, "스트림 없음: "+r.PathValue("id"), http.StatusNotFound)
			return
		}
		w.Header().Set("Content-Type", "application/json")
		json.NewEncoder(w).Encode(s)
	})

	// 처리 통계
	mux.HandleFunc("GET /stats", func(w http.ResponseWriter, r *http.Request) {
		streams, segments := st.Totals()
		w.Header().Set("Content-Type", "application/json")
		json.NewEncoder(w).Encode(map[string]int{
			"streams":       streams,
			"segments_done": segments,
		})
	})

	srv := &http.Server{Addr: ":8080", Handler: mux}

	// 서버는 별도 고루틴에서 — main 고루틴은 종료 신호를 기다린다.
	go func() {
		log.Println("게이트웨이 시작: http://localhost:8080")
		if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
			log.Fatalf("서버 오류: %v", err)
		}
	}()

	<-ctx.Done() // Ctrl-C 대기
	log.Println("종료 신호 수신 — graceful shutdown 시작")

	// 진행 중인 요청은 5초까지 기다렸다가 서버를 닫는다.
	shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	if err := srv.Shutdown(shutdownCtx); err != nil {
		log.Printf("서버 종료 오류: %v", err)
	}

	pool.Wait() // 워커들이 ctx.Done()을 보고 멈출 때까지 대기
	log.Println("모든 워커 종료 — 게이트웨이 정상 종료")
}

4. 확장 과제 — 답 없음, 힌트만

  • DELETE /streams/{id} — 힌트: store에 Delete 메서드 추가(맵 delete, 0004), 없는 ID면 404. 테스트도 테이블에 한 줄 추가.
  • 워커 재시도 — 힌트: Job에 Attempt int를 두고, "실패"(예: 세그먼트 번호가 3의 배수면 실패로 가정) 시 Attempt+1로 다시 Jobs에 넣기. 최대 횟수 초과 시 로그만. 닫힌 채널·꽉 찬 채널을 조심할 것 (0007).
  • 스트림별 통계 — 힌트: GET /streams/{id}/stats 라우트 추가, Stream에 처리 시간 합계 필드. 뮤텍스 임계구역을 넓히지 않고 할 수 있는지가 관전 포인트 (0012).
